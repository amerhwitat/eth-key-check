cmake_minimum_required(VERSION 3.21)
project(chimera_crypto_qt LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Widgets)
qt_standard_project_setup()
qt_add_executable(chimera_crypto_qt src/main.cpp)
target_link_libraries(chimera_crypto_qt PRIVATE Qt6::Core Qt6::Gui Qt6::Widgets)
set_target_properties(chimera_crypto_qt PROPERTIES WIN32_EXECUTABLE ON MACOSX_BUNDLE ON)
