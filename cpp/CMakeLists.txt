cmake_minimum_required(VERSION 3.21)
project(chimera_crypto_qt VERSION 1.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Widgets)
qt_standard_project_setup()
add_library(chimera_crypto_core INTERFACE)
target_include_directories(chimera_crypto_core INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/include)
qt_add_executable(chimera_crypto_qt src/main.cpp)
target_link_libraries(chimera_crypto_qt PRIVATE chimera_crypto_core Qt6::Core Qt6::Gui Qt6::Widgets)
add_executable(chimera_crypto_core_check tests/core_check.cpp)
target_link_libraries(chimera_crypto_core_check PRIVATE chimera_crypto_core)
set_target_properties(chimera_crypto_qt PROPERTIES WIN32_EXECUTABLE ON MACOSX_BUNDLE ON)
enable_testing()
add_test(NAME chimera_crypto_core_check COMMAND chimera_crypto_core_check)
